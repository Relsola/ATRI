fn sum6(a: i32, b: i32, c: i32, d: i32, e: i32, f: i32) -> i32 {
    return a + b + c + d + e + f;
}

fn divrem(x: i32, y: i32) -> i32 {
    q: i32 = x / y;
    r: i32 = x % y;
    return q * 10 + r;
}

fn wrap(x: i32, y: i32) -> i32 {
    return x + y;
}

fn bindings(x: i32) -> i32 {
    a: i32;
    a = x;
    a = a * 3;
    b: i32 = a + 1;
    a = 7;
    return b + a;
}

fn nested(x: i32, y: i32) -> i32 {
    return sum6(divrem(x, y), wrap(x, 1), wrap(y, 2), 3, 4, 5);
}

fn negate(x: i32) -> i32 {
    return -x;
}
